<script>{`
  ;(() => {
  const postSelector = '#post-container[data-post-slug="broad-strokes"] .markdown-content'
  const marker = String.fromCharCode(96)

  function getBroadStrokesContent() {
    return document.querySelector(postSelector)
  }

  function normalizeGoogleDocsMarkdown(text) {
    let output = String(text || '')
    const replacements = [
      ['\\\\#', '#'],
      ['\\\\*', '*'],
      ['\\\\[', '['],
      ['\\\\]', ']'],
      ['\\\\(', '('],
      ['\\\\)', ')'],
      ['\\\\|', '|'],
      ['\\\\-', '-'],
      ['\\\\~', '~'],
    ]

    replacements.forEach(([escaped, value]) => {
      output = output.split(escaped).join(value)
    })

    return output
      .split('\\\\' + marker)
      .join(marker)
      .replace(/[ \\t]+$/gm, '')
      .trim()
  }

  function normalizedElementText(element) {
    return normalizeGoogleDocsMarkdown(element.textContent || '')
  }

  function appendProgressMeter(parent, text) {
    const marks = text.replace(/[^█░]/g, '')
    const total = marks.length || 1
    const filled = marks.split('█').length - 1
    const percent = Math.max(0, Math.min(100, Math.round((filled / total) * 100)))
    const meter = document.createElement('span')
    const fill = document.createElement('span')

    meter.className = 'ainekio-progress-meter'
    meter.setAttribute('aria-label', percent + '% complete')
    meter.title = percent + '% complete'
    fill.className = 'ainekio-progress-fill'
    fill.style.width = percent + '%'
    meter.append(fill)
    parent.append(meter)
  }

  function appendPercentProgress(parent, text) {
    const sourceText = normalizeGoogleDocsMarkdown(text)
    const match = sourceText.match(/~?(\\d+(?:\\.\\d+)?)(?:\\s*-\\s*(\\d+(?:\\.\\d+)?))?%/)
    if (!match) {
      appendInlineMarkdown(parent, sourceText)
      return
    }

    const firstValue = Number(match[1])
    const secondValue = match[2] ? Number(match[2]) : firstValue
    const percent = Math.max(0, Math.min(100, Math.round((firstValue + secondValue) / 2)))
    const filled = Math.round(percent / 10)
    const meterText = '█'.repeat(filled) + '░'.repeat(Math.max(0, 10 - filled))
    const label = document.createElement('span')

    appendProgressMeter(parent, meterText)
    label.className = 'ainekio-progress-label'
    label.textContent = sourceText
    parent.append(label)
  }

  function appendInlineMarkdown(parent, text) {
    const sourceText = normalizeGoogleDocsMarkdown(text)
    const tokenPattern = new RegExp(
      marker + '([█░]{3,})' + marker + '|' + marker + '([^' + marker + ']+)' + marker + '|\\\\*\\\\*([^*]+)\\\\*\\\\*|\\\\[([^\\\\]]+)\\\\]\\\\((https?:\\\\/\\\\/[^\\\\s)]+|\\\\/[^\\\\s)]+)\\\\)|([█░]{3,})',
      'g',
    )
    let cursor = 0
    let match

    while ((match = tokenPattern.exec(sourceText)) !== null) {
      if (match.index > cursor) {
        parent.append(document.createTextNode(sourceText.slice(cursor, match.index)))
      }

      if (match[1]) {
        appendProgressMeter(parent, match[1])
      } else if (match[2]) {
        const code = document.createElement('code')
        code.textContent = match[2]
        parent.append(code)
      } else if (match[3]) {
        const strong = document.createElement('strong')
        strong.textContent = match[3]
        parent.append(strong)
      } else if (match[4] && match[5]) {
        const anchor = document.createElement('a')
        anchor.textContent = match[4]
        anchor.href = match[5]
        if (match[5].startsWith('http')) {
          anchor.rel = 'noopener noreferrer'
          anchor.target = '_blank'
        }
        parent.append(anchor)
      } else if (match[6]) {
        appendProgressMeter(parent, match[6])
      }

      cursor = tokenPattern.lastIndex
    }

    if (cursor < sourceText.length) {
      parent.append(document.createTextNode(sourceText.slice(cursor)))
    }
  }

  function splitTableRow(text) {
    return normalizeGoogleDocsMarkdown(text).replace(/^\\|/, '').replace(/\\|$/, '').split('|').map((cell) => cell.trim())
  }

  function isTableLine(element) {
    return element.tagName === 'P' && /^\\|.+\\|$/.test(normalizedElementText(element))
  }

  function isBlankParagraph(element) {
    return element.tagName === 'P' && normalizedElementText(element) === ''
  }

  function nextNonBlankSibling(element) {
    let sibling = element.nextElementSibling
    while (sibling && isBlankParagraph(sibling)) {
      sibling = sibling.nextElementSibling
    }

    return sibling
  }

  function isTableDivider(text) {
    return /^\\|?\\s*:?-{3,}:?\\s*(\\|\\s*:?-{3,}:?\\s*)+\\|?$/.test(normalizeGoogleDocsMarkdown(text))
  }

  function buildTable(rows) {
    const table = document.createElement('table')
    const thead = document.createElement('thead')
    const tbody = document.createElement('tbody')
    const headerRow = document.createElement('tr')
    const headers = splitTableRow(normalizedElementText(rows[0]))
    const tableClasses = ['ainekio-enhanced-table']

    if (headers.includes('Area') && headers.includes('Progress')) {
      tableClasses.push('ainekio-status-table')
    }

    table.className = tableClasses.join(' ')

    headers.forEach((cellText) => {
      const cell = document.createElement('th')
      appendInlineMarkdown(cell, cellText)
      headerRow.append(cell)
    })

    thead.append(headerRow)

    rows.slice(1).forEach((row) => {
      const text = normalizedElementText(row)
      if (isTableDivider(text)) return

      const bodyRow = document.createElement('tr')
      splitTableRow(text).forEach((cellText, cellIndex) => {
        const cell = document.createElement('td')
        cell.dataset.label = headers[cellIndex] || ''
        if (headers[cellIndex] === 'Progress' && !/[█░]{3,}/.test(cellText)) {
          appendPercentProgress(cell, cellText)
        } else {
          appendInlineMarkdown(cell, cellText)
        }
        bodyRow.append(cell)
      })
      tbody.append(bodyRow)
    })

    table.append(thead, tbody)
    return table
  }

  function enhanceTables(root) {
    root.querySelectorAll('p').forEach((current) => {
      if (!current.isConnected || !isTableLine(current)) return
      const previous = current.previousElementSibling
      if (previous && (isTableLine(previous) || isBlankParagraph(previous))) return

      const secondRow = nextNonBlankSibling(current)
      if (!secondRow || !isTableDivider(normalizedElementText(secondRow))) return

      const rows = []
      const spacerRows = []
      let cursor = current
      while (cursor) {
        const child = cursor
        if (isTableLine(child)) {
          rows.push(child)
          cursor = child.nextElementSibling
          continue
        }

        if (isBlankParagraph(child) && rows.length > 0 && isTableLine(nextNonBlankSibling(child))) {
          spacerRows.push(child)
          cursor = child.nextElementSibling
          continue
        }

        break
      }

      if (rows.length >= 3) {
        rows[0].replaceWith(buildTable(rows))
        rows.slice(1).forEach((row) => row.remove())
        spacerRows.forEach((row) => row.remove())
      }
    })
  }

  function enhanceNumberedLists(root) {
    const children = Array.from(root.children)
    for (let index = 0; index < children.length; index += 1) {
      const current = children[index]
      if (current.tagName !== 'P') continue

      const firstMatch = normalizedElementText(current).match(/^\\d+\\.\\s+(.+)$/)
      if (!firstMatch) continue

      const items = []
      let cursor = index
      while (cursor < children.length && children[cursor].tagName === 'P') {
        const match = normalizedElementText(children[cursor]).match(/^\\d+\\.\\s+(.+)$/)
        if (!match) break
        items.push({ element: children[cursor], text: match[1] })
        cursor += 1
      }

      if (items.length === 0) continue

      const list = document.createElement('ol')
      items.forEach((item) => {
        const listItem = document.createElement('li')
        appendInlineMarkdown(listItem, item.text)
        list.append(listItem)
      })

      items[0].element.replaceWith(list)
      items.slice(1).forEach((item) => item.element.remove())
      index = cursor - 1
    }
  }

  function enhanceBulletedLists(root) {
    const children = Array.from(root.children)
    for (let index = 0; index < children.length; index += 1) {
      const current = children[index]
      if (current.tagName !== 'P') continue

      const firstMatch = normalizedElementText(current).match(/^[-*]\\s+(.+)$/)
      if (!firstMatch) continue

      const items = []
      let cursor = index
      while (cursor < children.length && children[cursor].tagName === 'P') {
        const match = normalizedElementText(children[cursor]).match(/^[-*]\\s+(.+)$/)
        if (!match) break
        items.push({ element: children[cursor], text: match[1] })
        cursor += 1
      }

      const list = document.createElement('ul')
      items.forEach((item) => {
        const listItem = document.createElement('li')
        appendInlineMarkdown(listItem, item.text)
        list.append(listItem)
      })

      items[0].element.replaceWith(list)
      items.slice(1).forEach((item) => item.element.remove())
      index = cursor - 1
    }
  }

  function enhanceBlockquotes(root) {
    root.querySelectorAll('p').forEach((paragraph) => {
      const text = normalizedElementText(paragraph)
      if (!text.startsWith('> ')) return

      const blockquote = document.createElement('blockquote')
      appendInlineMarkdown(blockquote, text.slice(2))
      paragraph.replaceWith(blockquote)
    })
  }

  function enhanceHeadings(root) {
    root.querySelectorAll('p').forEach((paragraph) => {
      const text = normalizedElementText(paragraph)
      const boldHeading = text.match(/^\\*\\*(#{1,3})\\s+(.+?)\\*\\*$/)
      const plainHeading = text.match(/^(#{1,3})\\s+(.+)$/)
      const match = boldHeading || plainHeading
      if (!match) return

      const level = Math.min(match[1].length, 3)
      const heading = document.createElement('h' + level)
      heading.className = 'ainekio-doc-heading'
      appendInlineMarkdown(heading, match[2])
      paragraph.replaceWith(heading)
    })
  }

  function enhanceMetadata(root) {
    root.querySelectorAll('p').forEach((paragraph) => {
      if (paragraph.classList.contains('ainekio-verified-callout')) return
      const text = normalizedElementText(paragraph)
      if (!text.startsWith('Last verified:')) return

      paragraph.classList.add('ainekio-verified-callout')
      paragraph.textContent = ''
      appendInlineMarkdown(paragraph, text)
    })
  }

  function enhanceInlineMarkdown(root) {
    root.querySelectorAll('p, li').forEach((element) => {
      if (element.dataset.ainekioEnhanced === 'true') return
      const text = normalizedElementText(element)
      if (
        !text.includes('**') &&
        !text.includes('[') &&
        !text.includes(marker) &&
        !/[█░]{3,}/.test(text)
      ) {
        if (text !== element.textContent.trim()) {
          element.textContent = text
        }
        element.dataset.ainekioEnhanced = 'true'
        return
      }

      element.textContent = ''
      appendInlineMarkdown(element, text)
      element.dataset.ainekioEnhanced = 'true'
    })
  }

  function enhanceBroadStrokes() {
    const root = getBroadStrokesContent()
    if (!root) return

    enhanceHeadings(root)
    enhanceMetadata(root)
    enhanceTables(root)
    enhanceNumberedLists(root)
    enhanceBulletedLists(root)
    enhanceBlockquotes(root)
    enhanceInlineMarkdown(root)
  }

  const readyCheck = window.setInterval(() => {
    const root = getBroadStrokesContent()
    if (!root) return
    enhanceBroadStrokes()
    if (!root.textContent.includes('Loading document')) {
      window.clearInterval(readyCheck)
    }
  }, 250)

  const observer = new MutationObserver(() => enhanceBroadStrokes())
  const observerCheck = window.setInterval(() => {
    const root = getBroadStrokesContent()
    if (!root) return
    observer.observe(root, { childList: true, subtree: true })
    window.clearInterval(observerCheck)
  }, 250)
  })()
`}</script>
