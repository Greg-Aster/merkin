# Robot Builder Calculator

<div class="builder-draft-disclaimer">EXAMPLE ONLY. DRAFT. This calculator is a planning prototype, not a final purchase list or finished Ainekio hardware specification.</div>

Last researched: 2026-07-08

Ainekio does not have one correct brain. It has an architecture choice: borrow a large mind over the network, use a phone as the nearby brain, put a Linux service computer in the body, or pay the cost and power penalty for Jetson-class autonomy.

This calculator can start from a quick preset or from the guided brain choice, then filters the rest of the build around what is actually possible. Greyed-out options are not compatible with the selected brain tier. The cost bar is derived from rough real-world parts estimates, while the intelligence bar estimates useful autonomy rather than vibes.

## What The Tiers Mean

**Ainekio body/platform:** the final robot is Ainekio. A Sesame-style body is one possible reference base, not the product identity. The calculator now treats the body as a real tier: Sesame-style reference base, custom Ainekio body, or a popular TurtleBot-class mobile base.

**ESP32-S3 body controller:** an ESP32-S3-WROOM-1U-N16R8 style module can directly control servos if enough pins are available, or it can drive a PCA9685-style servo bus. Direct servo control is cheaper and tighter. A servo bus is cleaner and leaves more room for sensors.

**Seeed XIAO ESP32S3 Sense:** this is the tiny board with onboard camera and microphone. It can still directly control some servos or use a servo bus, but no-bus builds are pin-constrained because camera, mic, SD, I2C, and servo signals all compete for a tiny header.

**Zero-class Linux body:** Radxa Zero 3W plus a servo controller hat/bus is the clean one-board Linux version. It can do audio services, camera handling, bridge work, and light model fallback. A split version can add ESP32 for realtime motion/reflexes.

**Jetson Orin autonomy:** Orin Nano is the approachable local-AI tier, Orin NX is the stronger robot-brain tier, and AGX Orin-class systems are high-end edge AI. These are not tiny-body drop-in choices without serious power, cooling, and body changes.

    function clamp(value) {
      return Math.max(0, Math.min(100, Math.round(value)))
    }

    function createElement(tag, className, text) {
      const element = document.createElement(tag)
      if (className) element.className = className
      if (text) element.textContent = text
      return element

    function formatUsd(value) {
      return '$' + Math.round(value).toLocaleString('en-US')
    }

    function formatRange(min, max) {
      return formatUsd(min) + '-' + formatUsd(max)
    }

    function getQuestion(id) {
      return questions[id]
    }

    function getOption(questionId, optionId) {
      const question = getQuestion(questionId)
      if (!question) return null
      return question.options.find((option) => option.id === optionId) || null
    }

    function selectedBrain(state) {
      return state.answers.brain || ''
    }

    function selectedController(state) {
      return state.answers.controller || state.answers.zero || ''
    }

    function optionDisabledReason(option, state) {
      const brain = selectedBrain(state)
      const controller = selectedController(state)
      if (option.requiresBrain && !option.requiresBrain.includes(brain)) {
        return option.disabledReason || 'Not available for the selected brain.'
      }
      if (option.disallowBrain && option.disallowBrain.includes(brain)) {
        return option.disabledReason || 'Not available for the selected brain.'
      }
      if (option.requiresControllerPrefix && !controller.startsWith(option.requiresControllerPrefix)) {
        return option.disabledReason || 'Not available for the selected controller.'
      }
      return ''
    }

    function optionIsAllowed(option, state) {
      return optionDisabledReason(option, state) === ''
    }

    function nextQuestionId(questionId, option) {
      if (!option) return null
      return option.next || null
    }

    function activeQuestionIds(state) {
      const ids = []
      let questionId = 'brain'
      const guard = new Set()
      while (questionId && !guard.has(questionId)) {
        guard.add(questionId)
        ids.push(questionId)
        const option = getOption(questionId, state.answers[questionId])
        if (!option) break
        questionId = nextQuestionId(questionId, option)
      }
      return ids
    }

    function selectedOptions(state) {
      return activeQuestionIds(state)
        .map((questionId) => [questionId, getOption(questionId, state.answers[questionId])])
        .filter((entry) => entry[1] && optionIsAllowed(entry[1], state))
    }

    function applyPreset(root, state, preset) {
      state.answers = { ...preset.answers }
      state.currentId = null
      pruneAnswers(state)
      renderQuestion(root, state)
      renderOutput(root, state)
    }

    function renderPresets(root, state) {
      const target = root.querySelector('[data-builder-presets]')
      if (!target) return
      target.replaceChildren()
      presets.forEach((preset) => {
        const button = document.createElement('button')
        button.type = 'button'
        button.className = 'builder-preset'
        button.append(createElement('strong', '', preset.label))
        button.append(createElement('span', '', preset.summary))
        button.addEventListener('click', () => applyPreset(root, state, preset))
        target.append(button)
      })
    }

    function pruneAnswers(state) {
      const active = new Set(activeQuestionIds(state))
      Object.keys(state.answers).forEach((questionId) => {
        if (!active.has(questionId)) delete state.answers[questionId]
      })
    }

    function firstAllowedOption(questionId, state) {
      const question = getQuestion(questionId)
      return question.options.find((option) => optionIsAllowed(option, state)) || question.options[0]
    }

    function normalizeCurrent(state) {
      pruneAnswers(state)
      const current = getQuestion(state.currentId)
      if (!current) {
        state.currentId = null
        return
      }
      const selected = getOption(state.currentId, state.answers[state.currentId])
      if (selected && !optionIsAllowed(selected, state)) {
        delete state.answers[state.currentId]
      }
      if (state.currentId && !getQuestion(state.currentId)) {
        state.currentId = 'brain'
      }
    }

    function clampCostBar(max) {
      return clamp((max / 3800) * 100)
    }

    function scoreState(state) {
      const groups = {}
      let intelligence = 10
      selectedOptions(state).forEach(([questionId, option]) => {
        if (option.group) {
          groups[option.group] = {
            min: option.min || 0,
            max: option.max || 0,
            item: option.item || option.label,
            facts: option.facts,
          }
        }
        intelligence += option.intelligence || 0
      })
      const costItems = Object.values(groups)
      const estimate = costItems.reduce((total, item) => ({
        min: total.min + item.min,
        max: total.max + item.max,
      }), { min: 0, max: 0 })
      const selectedCount = selectedOptions(state).length || 1
      const profile = getProfile(state)
      return {
        profile,
        specs: getSpecs(state),
        estimate,
        cost: clampCostBar(estimate.max),
        intelligence: clamp(intelligence / selectedCount),
        costItems: costItems.map((item) => [item.item, formatRange(item.min, item.max)]),
        parts: getPartsList(state),
        facts: costItems.flatMap((item) => item.facts || []),
        selections: selectedOptions(state),
      }
    }

    function setCostMeter(root, score) {
      const fill = root.querySelector('[data-cost-fill]')
      const label = root.querySelector('[data-cost-value]')
      const track = root.querySelector('[data-cost-track]')
      if (!fill || !label || !track) return
      fill.style.width = score.cost + '%'
      label.textContent = formatRange(score.estimate.min, score.estimate.max)
      track.setAttribute('aria-valuenow', String(score.cost))
    }

    function setMeter(root, kind, value) {
      const fill = root.querySelector('[data-' + kind + '-fill]')
      const label = root.querySelector('[data-' + kind + '-value]')
      const track = root.querySelector('[data-' + kind + '-track]')

      fill.style.width = value + '%'
      label.textContent = value + '%'
      track.setAttribute('aria-valuenow', String(value))
    }

    }

    function renderQuestion(root, state) {
      const target = root.querySelector('[data-builder-questions]')
      if (!target) return
      normalizeCurrent(state)
      target.replaceChildren()
      if (!state.currentId) {
        const done = createElement('section', 'builder-question')
        done.append(createElement('h3', '', 'Build path complete'))
        done.append(createElement('p', '', 'Use Back to revise the branch, or Reset to start over.'))
        target.append(done)
        renderNav(target, root, state)
        renderOutput(root, state)
        return
      }
      const question = getQuestion(state.currentId)
      const section = createElement('section', 'builder-question')
      section.append(createElement('h3', '', question.title))
      section.append(createElement('p', '', question.intro))
      const options = createElement('div', 'builder-options')
      question.options.forEach((option) => {
        const disabledReason = optionDisabledReason(option, state)
        const button = document.createElement('button')
        button.type = 'button'
        button.className = 'builder-option'
        button.disabled = disabledReason !== ''
        button.setAttribute('aria-pressed', String(state.answers[state.currentId] === option.id))
        if (button.disabled) button.setAttribute('aria-disabled', 'true')
        const top = createElement('span', 'builder-option-top')
        top.append(createElement('strong', '', option.label))
        top.append(createElement('span', 'builder-option-cost', option.costLabel || formatRange(option.min || 0, option.max || 0)))
        const note = createElement('small', '', option.note)
        if (disabledReason) note.append(createElement('em', '', disabledReason))
        if (option.facts && option.facts.length > 0) {
          const facts = createElement('ul', 'builder-option-facts')
          option.facts.forEach((fact) => {
            facts.append(createElement('li', '', fact))
          })
          note.append(facts)
        }
        const wrap = document.createElement('span')
        wrap.append(top, note)
        button.append(wrap)
        button.addEventListener('click', () => {
          if (button.disabled) return
          state.answers[state.currentId] = option.id
          const next = nextQuestionId(state.currentId, option)
          state.currentId = next
          pruneAnswers(state)
          renderQuestion(root, state)
          renderOutput(root, state)
        })
        options.append(button)
      })
      section.append(options)
      target.append(section)
      renderNav(target, root, state)
      renderOutput(root, state)
    }

    function renderNav(target, root, state) {
      const nav = createElement('div', 'builder-nav')
